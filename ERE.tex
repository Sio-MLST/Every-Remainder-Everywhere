\begin{abstract}
We prove that for every modulus $N \ge 1$, every natural number $w$ not divisible by $3$, every residue $r$, and every threshold $B$, there is a natural number $n > B$ with $n \equiv r \pmod N$ whose Collatz trajectory reaches $w$. In particular, the set of predecessors of $w$ is infinite in every residue class modulo every $N$. The $2$-adic and $3$-adic parts of the modulus are handled by prescribing a parity prefix of the shortcut map. For the remaining factor $M$, coprime to $6$, we study the backward tree of $2w$ through a finite graph whose edges carry affine maps of $\mathbb{Z}/M\mathbb{Z}$. A commutator with an explicit closed walk shows that the group of labels of closed walks contains the translation $t\mapsto t+1$, which forces every residue to occur. All results have been verified in Lean~4 without external libraries.
\end{abstract}

\section{Introduction}\label{sec:intro}
Let $\C\colon\N\to\N$ be the Collatz map, $\C(n)=n/2$ for even $n$ and $\C(n)=3n+1$ for odd $n$. We write $n\reaches v$ if $\C^{(k)}(n)=v$ for some $k\ge 0$.

\begin{theorem}[Every Remainder Everywhere]\label{thm:main}
Let $N\ge 1$, let $w\in\N$ with $3\nmid w$, and let $r,B\in\N$. Then there exists $n\in\N$ with
\[
n>B,\qquad n\reaches w,\qquad n\equiv r \pmod N.
\]
\end{theorem}

\begin{corollary}\label{cor:infinite}
For every $N\ge 1$, every $w\in\N$ with $3\nmid w$ and every residue $r$ modulo $N$, the set $\{n\in\N : n\reaches w,\ n\equiv r \pmod N\}$ is infinite.
\end{corollary}
\begin{proof}
Apply Theorem~\ref{thm:main} with arbitrarily large $B$.
\end{proof}

The hypothesis $3\nmid w$ is necessary: if $3\mid w$, then no $m$ satisfies $3m+1=w$, so the only predecessor of $w$ is $2w$, which is again divisible by $3$; hence every $n\reaches w$ has the form $2^jw$ and is divisible by $3$.

\subsection*{Strategy}
Write $N=2^a3^bM$ with $\gcd(M,6)=1$.

\smallskip\noindent\emph{The factor $M$.} Predecessors of $w$ are generated by two backward steps, $n\mapsto 2n$ and $n\mapsto (n-1)/3$ (when this is an odd integer not divisible by $3$). Let $\Tr(2w)$ be the set they generate from $2w$. We record the residue modulo $M$ of a node together with its class modulo $3^{K+1}$ and its parity. These data determine a finite graph on which each backward step acts on residues modulo $M$ by an affine map $t\mapsto \lambda t+\beta$ with $\lambda$ a unit (Sections~\ref{sec:fibres}--\ref{sec:affine}). Labels of closed walks form a group $\Hol_M(K)$ of affine maps. If this group fixed a common residue, the common fixed point would produce a consistent assignment of residues to states, and we show in Section~\ref{sec:flat} that no such assignment exists. A commutator with an explicit closed walk, combined with reduction modulo divisors of $M$, turns this obstruction into the statement that $t\mapsto t+1$ lies in $\Hol_M(K)$ (Section~\ref{sec:translation}). Iterating this translation shows that every residue modulo $M$ occurs, in every $3$-adic cylinder (Section~\ref{sec:density}).

\smallskip\noindent\emph{The factor $2^a3^b$.} Searching backward modulo $2^a$ is awkward since doubling is nilpotent there. Instead we run the shortcut map forward along a prescribed parity prefix, which transfers the conditions modulo $2^a$ and $3^b$ to a congruence for a single tree node modulo $3^{j+b}M$ (Section~\ref{sec:all}).

\subsection*{Related work}
Expanding the iterates of the shortcut map along a parity prefix is the classical $2$-adic encoding of Collatz orbits due to Terras \cite{Terras76} and Everett \cite{Everett77}; Lagarias \cite{Lagarias85} surveys it together with the backward tree of predecessors. Applegate and Lagarias \cite{ApplegateLagarias95} use the tree of predecessors to obtain density bounds, and Wirsching's monograph \cite{Wirsching98} treats the parity-vector formalism in detail. Our forward reduction in Section~\ref{sec:all} is an instance of the parity-prefix expansion. The new ingredient is the analysis of the backward tree modulo $M$ in Sections~\ref{sec:fibres}--\ref{sec:density}, which proceeds by a group of affine maps attached to the tree rather than by counting or by a parity-vector formula. We are not aware of a previous statement of Theorem~\ref{thm:main} for arbitrary $N$.

\subsection*{Conventions}
Throughout, $N$ is a positive integer and $\N=\{0,1,2,\dots\}$. Since $3\mid0$, the condition $3\nmid w$ forces $w\ge1$. Since every residue class modulo $N$ contains a natural number, there is no loss of generality in taking $r\in\N$ in Theorem~\ref{thm:main}. For $n\in\Z$ and a modulus $M$ we write $(n:\ZM)$ for the class of $n$. We write $L_K=3^{K+1}$. An integer $x$ is \emph{live} if $3\nmid x$. The word \emph{unit} always refers to an invertible element of a residue ring. For $M$ coprime to $6$, the integers $2$ and $3$ are units of $\ZM$, and $2^{-1},3^{-1}$ denote their inverses.

\section{Backward Steps and the Tree}\label{sec:tree}
\begin{lemma}\label{lem:steps}
\leavevmode
\begin{enumerate}[label=(\arabic*), leftmargin=1.8em]
\item $\C(2a)=a$ and $\C^{(j)}(2^jm)=m$ for all $j,m\in\N$.
\item If $a\reaches b$ and $b\reaches c$ then $a\reaches c$.
\item $\C(m)=n$ if and only if $m=2n$, or $m$ is odd and $3m+1=n$.
\end{enumerate}
\end{lemma}
\begin{proof}
Immediate from the definition of $\C$.
\end{proof}

Thus $n$ has the predecessor $2n$, and the predecessor $(n-1)/3$ exactly when $n\equiv 4\pmod 6$, since then $n-1=6k+3$ and $(n-1)/3=2k+1$ is odd. We only use the second step when its result is live, that is, when $n\not\equiv 1\pmod 9$.

\begin{definition}[Backward tree]
For $v\in\N$, let $\Tr(v)\subseteq\N$ be the smallest set such that $v\in\Tr(v)$; if $n\in\Tr(v)$ then $2n\in\Tr(v)$ (step $D$); and if $n\in\Tr(v)$, $n\equiv 4\pmod 6$ and $\tfrac{n-1}{3}\not\equiv 0\pmod 3$, then $\tfrac{n-1}{3}\in\Tr(v)$ (step $U$).
\end{definition}

\begin{lemma}\label{lem:tree_basic}
Let $v\in\N$.
\begin{enumerate}[label=(\arabic*), leftmargin=1.8em]
\item Every $n\in\Tr(v)$ satisfies $n\reaches v$.
\item If $v$ is even and $n\in\Tr(v)$ is odd, then $n=(m-1)/3$ for some $m\in\Tr(v)$ with $m\equiv 4\pmod 6$ and $(m-1)/3\not\equiv0\pmod 3$.
\end{enumerate}
\end{lemma}
\begin{proof}
(1) By induction on the construction: $2n\reaches n$ in one step, and $\C\bigl(\tfrac{n-1}{3}\bigr)=n$ because $\tfrac{n-1}{3}$ is odd. (2) The seed is even and step $D$ yields even numbers, so an odd node arises from step $U$.
\end{proof}

From Section~\ref{sec:fibres} to Section~\ref{sec:density} the seed is $v$, an even integer; in Section~\ref{sec:all} it is $2w$. Note that $2w\reaches w$.

\section{Arithmetic Modulo \texorpdfstring{$3^{K+1}$}{3\string^(K+1)}}\label{sec:arith}

\begin{lemma}[Lifting the exponent for base $4$]\label{lem:lte}
For all $K,m\ge 0$,
\[
4^m\equiv 1\pmod{3^{K+1}}\iff 3^K\mid m.
\]
Moreover, for every $n\ge 0$ there is an integer $q$ with $4^{3^n}=1+3^{n+1}+3^{n+2}q$.
\end{lemma}
\begin{proof}
This is a special case of the lifting-the-exponent lemma; cf.\ Lucas~\cite{Lucas78}. We include a proof. The identity holds for $n=0$ ($4=1+3$). If $4^{3^n}=1+X$ with $X=3^{n+1}(1+3q)$, then $(1+X)^3=1+3X+3X^2+X^3$, where $3X=3^{n+2}+3^{n+3}q$ and $3X^2$, $X^3$ are multiples of $3^{n+3}$.

We prove the equivalence by induction on $K$. For $K=0$ it says $4^m\equiv1\pmod 3$, which always holds. Assume it for $K$. If $4^m\equiv1\pmod{3^{K+2}}$, then also modulo $3^{K+1}$, so $m=3^Kk$. By the identity, $4^{3^K}=1+3^{K+1}u$ with $3\nmid u$. Then $4^m=(1+3^{K+1}u)^k\equiv 1+k\,3^{K+1}u\pmod{3^{2K+2}}$, and $2K+2\ge K+2$, so $3^{K+2}\mid k\,3^{K+1}u$, hence $3\mid ku$ and $3\mid k$. Conversely, if $3^{K+1}\mid m$, write $m=3\cdot 3^Kk$. By induction $4^{3^Kk}=1+3^{K+1}t$, and $(1+3^{K+1}t)^3\equiv1\pmod{3^{K+2}}$.
\end{proof}

\begin{lemma}[$2$ generates the live residues]\label{lem:two_gen}
\leavevmode
\begin{enumerate}[label=(\arabic*), leftmargin=1.8em]
\item $2^m\equiv1\pmod{3^{K+1}}$ if and only if $2\cdot3^K\mid m$.
\item If $E_1,E_2<2\cdot3^K$ and $2^{E_1}\equiv2^{E_2}\pmod{3^{K+1}}$, then $E_1=E_2$.
\item For all live residues $c,d<3^{K+1}$ there is $j\ge0$ with $2^jc\equiv d\pmod{3^{K+1}}$.
\end{enumerate}
\end{lemma}
\begin{proof}
Part~(3) is the classical fact that $2$ is a primitive root modulo every power of $3$; we include the short proof. (1) If $2^m\equiv 1\pmod{3^{K+1}}$, reduction modulo $3$ forces $m$ even, say $m=2n$, so $4^n\equiv1$ and Lemma~\ref{lem:lte} gives $3^K\mid n$. The converse is Lemma~\ref{lem:lte} again. (2) Say $E_1\le E_2$. Cancel $2^{E_1}$, which is coprime to $3$, to get $2^{E_2-E_1}\equiv1$; by (1) $2\cdot 3^K\mid E_2-E_1<2\cdot3^K$. (3) The map $E\mapsto 2^Ec\bmod 3^{K+1}$ on $\{0,\dots,2\cdot3^K-1\}$ takes live values and is injective by (2), after cancelling the unit $c$. There are exactly $2\cdot3^K$ live residues below $3^{K+1}$, so it is surjective.
\end{proof}

\begin{lemma}\label{lem:four_cong}
For $K\ge1$: $4^{3^{K-1}}\equiv1+3^K\pmod{3^{K+1}}$.
\end{lemma}
\begin{proof}
This is the identity of Lemma~\ref{lem:lte} with $n=K-1$.
\end{proof}

\begin{lemma}\label{lem:four_power}
If $c<3^{K+1}$ is live and $c\equiv1\pmod 3$, then $c\equiv4^e\pmod{3^{K+1}}$ for some $e\ge0$.
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:two_gen}(3), $c\equiv2^j$ for some $j$. If $j$ were odd, then $2^j\equiv2\pmod 3$, contradicting $c\equiv1\pmod 3$. So $j=2e$ and $2^j=4^e$.
\end{proof}

\section{Fibres and Stabilization}\label{sec:fibres}
In Sections~\ref{sec:fibres}--\ref{sec:density}, $M\ge1$ is an integer coprime to $6$, and $v$ is an even integer. Then $2$ and $3$ are units of $\ZM$. (The case $M=1$, where $\ZM=0$, is allowed.)

\begin{definition}[Fibre]
For $K\ge0$, a live $c<3^{K+1}$ and $f\in\{0,1\}$, let
\[
\Fib Kcf=\bigl\{(n:\ZM) : n\in\Tr(v),\ n\equiv c\!\!\pmod{3^{K+1}},\ n\equiv f\!\!\pmod 2\bigr\}\subseteq\ZM.
\]
The pair $(c,f)$ is a \emph{state} of depth $K$.
\end{definition}

\begin{lemma}\label{lem:fibre_basic}
\leavevmode
\begin{enumerate}[label=(\arabic*), leftmargin=1.8em]
\item (Projection) For $J\le K$, $\Fib Kcf\subseteq\Fib J{c\bmod 3^{J+1}}f$.
\item (Doubling) $t\in\Fib Kcf\Rightarrow 2t\in\Fib K{2c\bmod3^{K+1}}0$. Hence $|\Fib Kcf|\le|\Fib K{2c\bmod 3^{K+1}}0|$, as $t\mapsto2t$ is injective on $\ZM$.
\item (Equal sizes) $|\Fib Kc0|=|\Fib Kd0|$ for all live $c,d<3^{K+1}$.
\item (Exact doubling) $\Fib K{2c\bmod 3^{K+1}}0=2\cdot\Fib Kc0$.
\item (Parity inclusion) $\Fib Kc1\subseteq\Fib Kc0$.
\item (Undo correspondence) For live $t_0<3^{K+1}$ and $t\in\ZM$,
\[
t\in\Fib K{t_0}1\iff 3t+1\in\Fib{K+1}{(3t_0+1)\bmod3^{K+2}}0.
\]
\end{enumerate}
\end{lemma}
\begin{proof}
(1) $3^{J+1}\mid3^{K+1}$. (2) If $n\in\Tr(v)$ realizes $t$, then $2n\in\Tr(v)$ realizes $2t$ and is even. (3) By Lemma~\ref{lem:two_gen}(3), $d\equiv2^jc$ and $c\equiv2^id$; iterating (2) gives $|\Fib Kc0|\le|\Fib Kd0|\le|\Fib Kc0|$. (4) By (2) the map $t\mapsto2t$ embeds $\Fib Kc0$ in $\Fib K{2c}0$, and the sizes agree by (3). (5) If $t\in\Fib Kc1$ then $2t\in\Fib K{2c}0=2\cdot\Fib Kc0$ by (2), (4); cancel the unit $2$. (6) ($\Rightarrow$) An odd $n\in\Tr(v)$ in class $t_0$ satisfies $n=(m-1)/3$ with $m=3n+1\in\Tr(v)$ by Lemma~\ref{lem:tree_basic}(2); then $m\equiv3t_0+1\pmod{3^{K+2}}$ and $(m:\ZM)=3t+1$, and $m$ is even. ($\Leftarrow$) Let $m\in\Tr(v)$ be even with $m\equiv3t_0+1\pmod{3^{K+2}}$ and $(m:\ZM)=3t+1$. Then $m\equiv1\pmod3$ and $m\not\equiv1\pmod 9$, since $t_0$ is live; with $m$ even, $m\equiv4\pmod 6$. So step $U$ yields the odd node $n=(m-1)/3\in\Tr(v)$ with $n\equiv t_0\pmod{3^{K+1}}$ and $(n:\ZM)=t$, where we cancel the unit $3$ in $\ZM$.
\end{proof}

\begin{definition}
Let $\sigma_K=|\Fib K10|$.
\end{definition}

\begin{lemma}\label{lem:sigma}
\leavevmode
\begin{enumerate}[label=(\arabic*), leftmargin=1.8em]
\item $\sigma_{K+1}\le\sigma_K$.
\item $|\Fib K{t_0}1|=\sigma_{K+1}$ for every live $t_0<3^{K+1}$.
\end{enumerate}
\end{lemma}
\begin{proof}
(1) Lemma~\ref{lem:fibre_basic}(1) with the class $1$. (2) By Lemma~\ref{lem:fibre_basic}(6), $\Fib K{t_0}1$ is the preimage of $\Fib{K+1}{3t_0+1}0$ under the bijection $t\mapsto3t+1$, and by Lemma~\ref{lem:fibre_basic}(3) this fibre has size $\sigma_{K+1}$.
\end{proof}

\begin{proposition}[Stabilization]\label{prop:stab}
There is $K_0\ge0$ such that for all $K\ge K_0$:
\begin{enumerate}[label=(\alph*), leftmargin=1.8em]
\item $\Fib Kc1=\Fib Kc0$ for every live $c<3^{K+1}$;
\item $\Fib{K+1}e0=\Fib K{e\bmod3^{K+1}}0$ for every live $e<3^{K+2}$.
\end{enumerate}
\end{proposition}
\begin{proof}
A non-increasing sequence of non-negative integers is eventually constant; choose $K_0$ with $\sigma_K=\sigma_{K_0}$ for $K\ge K_0$. (a) $\Fib Kc1\subseteq\Fib Kc0$ by Lemma~\ref{lem:fibre_basic}(5); their sizes are $\sigma_{K+1}$ and $\sigma_K$, which agree. (b) $\Fib{K+1}e0\subseteq\Fib K{e\bmod3^{K+1}}0$ by projection, and both have size $\sigma_{K+1}=\sigma_K$.
\end{proof}

\section{Affine Labels, Walks and Holonomy}\label{sec:affine}
Let $\Aff(M)=(\ZM)^\times\times\ZM$, where $f=(f_1,f_2)$ acts on $\ZM$ by $f(t)=f_1t+f_2$. Composition is written diagrammatically: $f\then g=(g_1f_1,\;g_1f_2+g_2)$ means ``first $f$, then $g$''. The identity is $(1,0)$, and $f^{-1}=(f_1^{-1},-f_1^{-1}f_2)$.

\begin{definition}[Edges, walks, holonomy]\label{def:edges}
Fix a depth $K\ge0$. The vertices are the states $(c,f)$. There are two kinds of directed edges.
\begin{itemize}
\item \textbf{$D$-edges.} For live $c<3^{K+1}$ and $f\in\{0,1\}$: $(c,f)\to(2c\bmod3^{K+1},0)$ with label $(2,0)$, that is, $t\mapsto2t$.
\item \textbf{$U$-edges.} For $e<3^{K+2}$ with $e\equiv1\pmod3$ and $e\not\equiv1\pmod9$: $(e\bmod3^{K+1},0)\to\bigl(\tfrac{e-1}{3}\bmod3^{K+1},1\bigr)$ with label $(3^{-1},-3^{-1})$, that is, $t\mapsto(t-1)/3$.
\end{itemize}
A \emph{walk} is a finite sequence of edges, each traversed forward or backward; its label is the composition of the labels, with the inverse label for a backward traversal. Let $\Hol_M(K)\le\Aff(M)$ be the group of labels of closed walks at the base state $(1,0)$.
\end{definition}

\begin{remark}
The $U$-edge requires no explicit parity condition on $e$. Its source has parity flag $0$, so the nodes of the corresponding fibre are even; together with $e\equiv1\pmod 3$ this gives $n\equiv4\pmod6$, which is what step $U$ requires. The condition $e\not\equiv1\pmod 9$ makes the target live.

The states and edges with their labels form a gain graph in the sense of Zaslavsky \cite{Zaslavsky89}, also called a voltage graph \cite{GrossTucker87}, with gains in $\Aff(M)$ acting on $\ZM$. In that language $\Hol_M(K)$ is the group of gains of closed walks at $(1,0)$; we keep the word \emph{holonomy} for its geometric suggestiveness.
\end{remark}

\begin{proposition}[Labels act on fibres]\label{prop:walk_image}
Let $K\ge K_0$. For every edge $u\to u'$ with label $f$, $f(F_K(u))=F_K(u')$. Consequently, for every walk from $u$ to $u'$ with label $g$, $g(F_K(u))=F_K(u')$. In particular, each $h\in\Hol_M(K)$ maps $\Fib K10$ bijectively onto itself.
\end{proposition}
\begin{proof}
For a $D$-edge, $\Fib Kc{f_0}=\Fib Kc0$ by Proposition~\ref{prop:stab}(a), and $2\cdot\Fib Kc0=\Fib K{2c}0$ by Lemma~\ref{lem:fibre_basic}(4). For a $U$-edge, $\Fib K{e\bmod3^{K+1}}0=\Fib{K+1}e0$ by Proposition~\ref{prop:stab}(b), and Lemma~\ref{lem:fibre_basic}(6) carries this onto $\Fib K{(e-1)/3}1$ via $t\mapsto(t-1)/3$. Backward traversals invert a bijection. The statement for walks follows by induction on length.
\end{proof}

\begin{lemma}[Connectedness]\label{lem:connected}
For every $K\ge0$, every state $(c,f)$ is joined to $(1,0)$ by a walk.
\end{lemma}
\begin{proof}
For $(c,0)$, Lemma~\ref{lem:two_gen}(3) gives $j$ with $2^j\equiv c\pmod{3^{K+1}}$, and $j$ successive $D$-edges lead from $(1,0)$ to $(c,0)$. For $(c,1)$, put $e=3c+1<3^{K+2}$. Then $e\equiv1\pmod3$ and $e\not\equiv1\pmod9$ because $c$ is live, and a $U$-edge joins $(e\bmod3^{K+1},0)$ to $(c,1)$.
\end{proof}

\section{The Modular Obstruction to Flat Assignments}\label{sec:flat}
\begin{definition}[Flat assignment]\label{def:flat}
Let $M\ge1$ be coprime to $6$ and $K\ge1$. A function $\Phi$ from states of depth $K$ to $\ZM$ is \emph{flat} if $\Phi(u')=f(\Phi(u))$ for every edge $u\to u'$ with label $f$, that is,
\begin{equation}\label{eq:flat}
\Phi(2c\bmod L_K,0)=2\Phi(c,f),\qquad 3\,\Phi\bigl(\tfrac{e-1}{3}\bmod L_K,1\bigr)=\Phi(e\bmod L_K,0)-1 .
\end{equation}
\end{definition}

\begin{theorem}[No flat assignment]\label{thm:noflat}
Let $M>1$ be coprime to $6$ and $K\ge1$. There is no flat assignment at depth $K$.
\end{theorem}
\begin{proof}
Suppose $\Phi$ is flat, and put $\phi(c)=\Phi(c,0)$. We use only that $2$ and $3$ are units of $\ZM$ and that $\ZM\neq0$.

\medskip\noindent\textbf{Step 1 (collapse and normalization).} The two $D$-edges out of $(c,0)$ and $(c,1)$ give $2\Phi(c,0)=2\Phi(c,1)$, so $\Phi(c,0)=\Phi(c,1)$, and $\phi(2c)=2\phi(c)$. The $U$-edge with $e=4$ gives $3\Phi(1,1)=\phi(4)-1$. Since $\phi(4)=4\phi(1)$ and $\Phi(1,1)=\phi(1)$, we get $3\phi(1)=4\phi(1)-1$, so $\phi(1)=1$.

\medskip\noindent\textbf{Step 2 (powers of $2$ and $4$).} By induction, $\phi(2^j\bmod L_K)=2^j$ in $\ZM$ for all $j\ge0$. If $c$ is live, $c\equiv1\pmod3$, $c<L_K$ and $c\equiv4^e\pmod{L_K}$ (Lemma~\ref{lem:four_power}), then $\phi(c)=4^e$.

\medskip\noindent\textbf{Step 3 (self-targeting).} Write $K=n+1$ and $s=3^n$, so $L_K=9s$ and $3^K=3s$. Let $y=(9s-1)/2$. Then $y<9s$, $y\equiv1\pmod 3$ and $y\equiv4\pmod9$. The numbers $e_1=y+9s$ and $e_2=y+18s$ are less than $27s=3^{K+2}$, are $\equiv4\pmod9$, and both reduce to $y$ modulo $9s$. Since $9s-1=2y$, their undo targets are
\[
\tfrac{e_1-1}{3}=y,\qquad\tfrac{e_2-1}{3}=y+3s .
\]
The second relation in \eqref{eq:flat} for $e_1$ and $e_2$ gives
\[
3\Phi(y,1)=\phi(y)-1=3\Phi(y+3s,1).
\]
Cancelling the unit $3$ and using collapse, $\phi(y+3s)=\phi(y)$.

Choose $j$ with $2^j\equiv y\pmod{9s}$. By Lemma~\ref{lem:four_cong}, $4^s\equiv1+3s\pmod{9s}$, hence
\[
2^{j+2s}=2^j4^s\equiv y(1+3s)\equiv y+3s\pmod{9s},
\]
because $3sy\equiv3s\pmod{9s}$ as $y\equiv1\pmod 3$. By Step~2, $\phi(y)=2^j$ and $\phi(y+3s)=2^{j+2s}$. Thus $2^{j+2s}=2^j$ in $\ZM$, and cancelling the unit $2^j$ gives
\begin{equation}\label{eq:four}
4^{3^{K-1}}=1\quad\text{in }\ZM .
\end{equation}

\medskip\noindent\textbf{Step 4 (the order of $4$).} The set $I=\{m\in\Z:4^m=1\text{ in }\ZM\}$ is a subgroup of $\Z$, since $4$ is a unit of $\ZM$. By \eqref{eq:four} it contains $3^{K-1}>0$, so $I=d\Z$ for the multiplicative order $d\ge 1$ of $4$, and $d\mid3^{K-1}$. Hence $d=3^a$ with $0\le a\le K-1$, and
\begin{equation}\label{eq:order}
4^m=1\text{ in }\ZM\iff3^a\mid m\qquad(m\ge0).
\end{equation}
If $a=0$ then $4=1$ in $\ZM$, so $3=0$, which is impossible since $3$ is a unit and $\ZM\ne0$. Hence $1\le a\le K-1$.

\medskip\noindent\textbf{Step 5 (two coupled nodes).} Write $a=a'+1$ and $r=3^{a'}$, so $3^a=3r$ and $3^{a+1}=9r\le3^K$. Let $c=4+9r$. Then $c<L_K$, $c\equiv4\pmod 9$, so $c$ is the source of a $U$-edge with target $(c-1)/3=1+3r$.

Write $c\equiv4^{e}\pmod{L_K}$ (Lemma~\ref{lem:four_power}); $e\ge1$ because $c\not\equiv1$. Write $e=e'+1$. Reducing modulo $9r$ gives $4^{e'+1}\equiv4$, so $4^{e'}\equiv1\pmod{3^{a+1}}$; Lemma~\ref{lem:lte} gives $3^a\mid e'$, and \eqref{eq:order} gives $4^{e'}=1$ in $\ZM$. Thus $\phi(c)=4^{e'+1}=4$. The $U$-edge relation at $c$ reads $3\Phi(1+3r,1)=\phi(c)-1=3$, so $\phi(1+3r)=1$.

On the other hand, $1+3r\equiv1\pmod 3$ and Lemma~\ref{lem:four_power} gives $1+3r\equiv4^{\delta}\pmod{L_K}$, so $\phi(1+3r)=4^{\delta}$. Hence $4^{\delta}=1$ in $\ZM$, so $3^a\mid\delta$ by \eqref{eq:order}, and Lemma~\ref{lem:lte} gives $4^{\delta}\equiv1\pmod{3^{a+1}}$, that is, modulo $9r$. But $4^{\delta}\equiv1+3r\pmod{9r}$ and $0<3r<9r$, so $1+3r\not\equiv1\pmod{9r}$. This is a contradiction.
\end{proof}

\section{From the Obstruction to the Translation \texorpdfstring{$t\mapsto t+1$}{t -> t+1}}\label{sec:translation}

\begin{remark}\label{rem:z55}
Over a ring such as $\Z/55\Z$, a group of affine maps without common fixed point need not contain a non-zero translation: $h(t)=36t+11$ generates a cyclic group of order $5$ with no fixed point and no non-zero translation, as $1-36=-35$ is a zero divisor. We therefore do not derive a translation from the absence of a fixed point directly. Instead we exhibit an explicit closed walk whose slope $s$ satisfies that $s-1$ is a unit, and use it to compute commutators.
\end{remark}

\begin{lemma}[A common fixed point gives a flat assignment]\label{lem:fixed_flat}
Let $K\ge1$. If $x\in\ZM$ satisfies $h(x)=x$ for all $h\in\Hol_M(K)$, then a flat assignment exists at depth $K$.
\end{lemma}
\begin{proof}
For each state $u$ choose, by Lemma~\ref{lem:connected}, a walk from $(1,0)$ to $u$ with label $g_u$, and set $\Phi(u)=g_u(x)$. If $g,g'$ are labels of two walks from $(1,0)$ to $u$, then $g\then g'^{-1}\in\Hol_M(K)$ fixes $x$, so $g(x)=g'(x)$; thus $\Phi(u)$ does not depend on the choice. For an edge $u\to u'$ with label $f$, appending this edge to the chosen walk to $u$ gives a walk to $u'$ with label $g_u\then f$, so $\Phi(u')=f(g_u(x))=f(\Phi(u))$.
\end{proof}

\begin{lemma}[The loop $w_0$]\label{lem:w0}
Let $K\ge1$. The closed walk
\[
w_0=D\then D\then U\then D\then D^{-1}
\]
at $(1,0)$ consists of the edges
\[
(1,0)\xrightarrow{\;D\;}(2,0)\xrightarrow{\;D\;}(4,0)\xrightarrow{\;U\,(e=4)\;}(1,1)\xrightarrow{\;D\;}(2,0),
\]
followed by the first edge $(1,0)\to(2,0)$ traversed backwards. Its label is
\[
w_0(t)=1+\tfrac43(t-1),\qquad\text{i.e. } w_0=(4\cdot3^{-1},\,-3^{-1}).
\]
\end{lemma}
\begin{proof}
All edges exist at every depth $K\ge1$ since $4<3^{K+1}$, $4\equiv1\pmod 3$, $4\not\equiv1\pmod 9$ and $(4-1)/3=1$. The label is obtained by following $t\mapsto2t\mapsto4t\mapsto\frac{4t-1}{3}\mapsto\frac{2(4t-1)}{3}\mapsto\frac{4t-1}{3}$, and $\frac{4t-1}{3}=1+\frac43(t-1)$.
\end{proof}

Thus $w_0$ fixes $1$, and its slope $s=4\cdot3^{-1}$ satisfies $s-1=3^{-1}$, a unit.

\begin{lemma}[Commutators with $w_0$]\label{lem:comm}
Let $K\ge1$ and $k=(\ell,m)\in\Hol_M(K)$. Then $c=k\then w_0\then k^{-1}\then w_0^{-1}$ is a translation $(1,\tau)\in\Hol_M(K)$ with $\tau=c(1)-1$, and
\[
k(1)=1+4\ell\,\tau .
\]
\end{lemma}
\begin{proof}
The slope of $c$ is $\ell s\ell^{-1}s^{-1}=1$, so $c=(1,\tau)$ with $\tau=c(1)-1$; and $c\in\Hol_M(K)$ as a product of elements of the group. Put $a=k(1)$ and write the maps centered at $1$:
\[
\begin{gathered}
k(t)=a+\ell(t-1),\qquad w_0(t)=1+s(t-1),\\
k^{-1}(u)=1+\ell^{-1}(u-a),\qquad w_0^{-1}(u)=1+s^{-1}(u-1).
\end{gathered}
\]
Then $k$ sends $1$ to $a$, $w_0$ sends $a$ to $1+s(a-1)$, $k^{-1}$ sends this to $1+\ell^{-1}\bigl((1-a)+s(a-1)\bigr)=1+\ell^{-1}(s-1)(a-1)$, and $w_0^{-1}$ sends this to
\[
c(1)=1+s^{-1}\ell^{-1}(s-1)(a-1).
\]
Since $s-1=3^{-1}$ and $s=4\cdot3^{-1}$, we have $s^{-1}(s-1)=4^{-1}$ (here $4$ is a unit as $2$ is), so $\tau=\ell^{-1}4^{-1}(a-1)$. Hence $a=1+4\ell\tau$.
\end{proof}

\begin{lemma}[Lifting walks modulo a divisor]\label{lem:lift}
Let $q\mid M$ and let $\rho\colon\ZM\to\Zq$ be reduction. Extend $\rho$ to $\Aff(M)\to\Aff(q)$ componentwise. For every $h'\in\Hol_q(K)$ there is $h\in\Hol_M(K)$ with $\rho(h)=h'$.
\end{lemma}
\begin{proof}
The graph of states and the sets of edges do not depend on the modulus; only the labels do, and the label of each edge modulo $q$ is the reduction of its label modulo $M$, because $\rho$ is a ring homomorphism and therefore $\rho(3^{-1}_M)=3^{-1}_q$. Moreover $\rho$ is compatible with composition and inversion of labels. Replacing each edge of a walk modulo $q$ by the same edge modulo $M$ gives a walk modulo $M$ with the required label.
\end{proof}

\begin{theorem}[The translation by $1$]\label{thm:translation}
Let $M\ge1$ be coprime to $6$ and $K\ge1$. Then $(1,1)\in\Hol_M(K)$; that is, $t\mapsto t+1$ is the label of a closed walk at $(1,0)$.
\end{theorem}
\begin{proof}
The set $\mathcal S=\{d\in\Z:(1,d\bmod M)\in\Hol_M(K)\}$ is a subgroup of $\Z$, since $\Hol_M(K)$ is a group and $(1,d_1)\then(1,d_2)=(1,d_1+d_2)$. It contains $M$, because $(1,0)$ is the label of the empty walk. Hence $\mathcal S=q\Z$, where $q$ is the least positive element of $\mathcal S$, and $q\mid M$. We show $q=1$.

If $M=1$ there is nothing to prove, so suppose $q>1$. Then $q\mid M$ is coprime to $6$, and $\Zq\ne0$. Let $h'\in\Hol_q(K)$. By Lemma~\ref{lem:lift} there is $h=(\ell,m)\in\Hol_M(K)$ with $\rho(h)=h'$, and by Lemma~\ref{lem:comm} there is $\tau$ with $(1,\tau)\in\Hol_M(K)$ and $h(1)=1+4\ell\tau$. An integer representative of $\tau$ lies in $\mathcal S=q\Z$, so $\rho(\tau)=0$, and therefore
\[
h'(1)=\rho(h(1))=1+4\rho(\ell)\rho(\tau)=1 .
\]
So $1\in\Zq$ is a common fixed point of $\Hol_q(K)$. By Lemma~\ref{lem:fixed_flat}, there is a flat assignment modulo $q$ at depth $K$, contradicting Theorem~\ref{thm:noflat} (note $q>1$). Hence $q=1$, so $1\in\mathcal S$.
\end{proof}

\section{Simultaneous Modular Density}\label{sec:density}

\begin{theorem}[Simultaneous density]\label{thm:density}
Let $M\ge1$ be coprime to $6$, let $v$ be an even integer with $3\nmid v$, let $J\ge0$, let $c<3^{J+1}$ be live and let $r\in\ZM$. Then there is $n\in\Tr(v)$ with
\[
n\equiv c\pmod{3^{J+1}}\qquad\text{and}\qquad(n:\ZM)=r .
\]
\end{theorem}
\begin{proof}
Let $K_0$ be as in Proposition~\ref{prop:stab} and $K=\max(K_0,J,1)$. The seed $v$ is an even, live node of $\Tr(v)$, so $\Fib K{v\bmod3^{K+1}}0\ne\varnothing$, and by Lemma~\ref{lem:fibre_basic}(3) the fibre $\Fib K10$ is non-empty. By Theorem~\ref{thm:translation}, $(1,1)\in\Hol_M(K)$, so by Proposition~\ref{prop:walk_image}, $t\in\Fib K10\Rightarrow t+1\in\Fib K10$. Starting from any element, repeated addition of $1$ reaches every residue, so $\Fib K10=\ZM$. By Lemma~\ref{lem:fibre_basic}(3), $|\Fib K{c'}0|=|\ZM|$ for every live $c'<3^{K+1}$, so $\Fib K{c'}0=\ZM$. Projecting from depth $K$ to depth $J$ (Lemma~\ref{lem:fibre_basic}(1)) with $c'=c$ gives $r\in\Fib Jc0$.
\end{proof}

\begin{remark}
For $M=1$ the congruence modulo $M$ is vacuous, and the theorem says that doubling alone reaches every $3$-adic cylinder: by Lemma~\ref{lem:two_gen}(3) some $2^jv$ lies in any prescribed live class modulo $3^{J+1}$. The argument above covers this case uniformly, since then $\Hol_M(K)$ is trivial and $q=M=1$.
\end{remark}

\section{Arbitrary Moduli via Forward Parity Prefixes}\label{sec:all}
Let $N=2^a3^bM$ with $\gcd(M,6)=1$. We prescribe the first $a'$ steps of the shortcut map forward.

\subsection*{The shortcut map and parity prefixes}
Let $T(n)=n/2$ for even $n$ and $T(n)=(3n+1)/2$ for odd $n$. For $a\ge0$ and $r\in\N$, let $j(a,r)$ be the number of odd integers among $r,T(r),\dots,T^{a-1}(r)$.

\begin{lemma}\label{lem:shortcut}
For all $n\in\N$ and $a\ge0$, $n\reaches T(n)$ and hence $n\reaches T^a(n)$.
\end{lemma}
\begin{proof}
If $n$ is even, $\C(n)=n/2=T(n)$. If $n$ is odd, $\C(n)=3n+1$ is even and $\C(3n+1)=(3n+1)/2=T(n)$. Use transitivity.
\end{proof}

\begin{lemma}[Parity prefix expansion]\label{lem:parity}
For all $a,r,t\ge0$,
\[
T^a(r+2^at)=T^a(r)+3^{j(a,r)}t .
\]
\end{lemma}
\begin{proof}
Induction on $a$; the case $a=0$ is trivial. For the step, $r$ and $r+2^{a+1}t$ have the same parity $\varepsilon$, and $T(r+2^{a+1}t)=T(r)+2^a3^{\varepsilon}t$. Apply the hypothesis to $T(r)$ at level $a$ and note $j(a+1,r)=\varepsilon+j(a,T(r))$.
\end{proof}

\begin{lemma}\label{lem:live_after_odd}
If $j(a,r)\ge1$ then $3\nmid T^a(r)$.
\end{lemma}
\begin{proof}
If $x$ is odd then $2T(x)=3x+1\equiv1\pmod3$, so $3\nmid T(x)$. If $x$ is even and $3\nmid x$, then $3\nmid x/2$. Hence once an odd step has occurred, every later term is live; induct on $a$.
\end{proof}

\begin{lemma}[An odd step in the prefix]\label{lem:normalize}
For all $a,r\ge0$ there exist $a'\ge a$ and $r'\equiv r\pmod{2^a}$ with $j(a',r')\ge1$.
\end{lemma}
\begin{proof}
Let $r_0=r\bmod 2^a$ and $j_0=j(a,r_0)$. Since $3^{j_0}$ is odd, there is $t\in\{0,1\}$ with $T^a(r_0)+3^{j_0}t$ odd. Take $a'=a+1$ and $r'=r_0+2^at$. By Lemma~\ref{lem:parity}, $T^a(r')$ is odd, so $j(a',r')\ge1$.
\end{proof}

\subsection*{Density input}
\begin{lemma}[Deep lift of a live class]\label{lem:lift_class}
Let $k\ge1$, $B\ge0$ and $c$ live. There exist $K\ge k$ and a live $d<3^{K+1}$ with $d\equiv c\pmod{3^k}$ and $d>B$.
\end{lemma}
\begin{proof}
Take $K=k+B+1$ and $d=(c\bmod3^k)+3^k(B+1)$. Then $d\equiv c\pmod{3^k}$, so $d$ is live; $d>B$; and $d<3^k+3^k(B+1)=3^k(B+2)<3^{K+1}$ since $B+2<3^{B+2}$.
\end{proof}

\begin{proposition}[Density input]\label{prop:density_input}
Let $M\ge1$ be coprime to $6$, let $w\in\N$ with $3\nmid w$, $k\ge1$, $c$ live, $\mu\in\N$ and $B\in\N$. There is $m\in\Tr(2w)$ with
\[
m\equiv c\pmod{3^k},\qquad m\equiv\mu\pmod M,\qquad m>B .
\]
\end{proposition}
\begin{proof}
Take $K,d$ from Lemma~\ref{lem:lift_class}. Theorem~\ref{thm:density} with seed $2w$ (even, live), depth $J=K$, class $d$ and residue $(\mu:\ZM)$ gives $m\in\Tr(2w)$ with $m\equiv d\pmod{3^{K+1}}$ and $m\equiv\mu\pmod M$. Since $m\equiv d\pmod{3^{K+1}}$ and $d<3^{K+1}$, we have $m\ge d>B$, and $m\equiv d\equiv c\pmod{3^k}$.
\end{proof}

\subsection*{Arithmetic}
\begin{lemma}\label{lem:arith}
\leavevmode
\begin{enumerate}[label=(\alph*), leftmargin=1.8em]
\item Every $N\ge1$ is of the form $2^a3^bM$ with $\gcd(M,6)=1$.
\item If $\gcd(P,Q)=1$, then ($n\equiv r\bmod P$ and $n\equiv r\bmod Q$) $\iff n\equiv r\pmod{PQ}$.
\item If $Q$ is odd, then for all $a,r',r$ there is $s$ with $r'+2^as\equiv r\pmod Q$.
\item If $e+3^jt\equiv e+3^js\pmod{3^jQ}$ then $t\equiv s\pmod Q$. If $\gcd(M,3)=1$ and $e+3^jt\equiv e+3^js\pmod M$ then $t\equiv s\pmod M$.
\end{enumerate}
\end{lemma}
\begin{proof}
(a) Divide out $2$ and $3$. (b) Chinese remainder theorem. (c) $2^a$ is a unit modulo $Q$; take $s\equiv2^{-a}(r-r')$. (d) Cancel $e$, then divide by $3^j$ in the first case; in the second, $3^j$ is a unit modulo $M$.
\end{proof}

\subsection*{The forward construction}
\begin{proposition}[Forward parity construction]\label{prop:construct}
Let $M\ge1$ be coprime to $6$, let $w\in\N$ with $3\nmid w$, let $a',r'\ge0$ with $j=j(a',r')\ge1$, and let $b,s,B\ge0$. There is $n>B$ with $n\reaches w$,
\[
n\equiv r'\pmod{2^{a'}}\qquad\text{and}\qquad n\equiv r'+2^{a'}s\pmod{3^bM}.
\]
\end{proposition}
\begin{proof}
Let $e=T^{a'}(r')$. It is live by Lemma~\ref{lem:live_after_odd}, so $e+3^js$ is live. Apply Proposition~\ref{prop:density_input} with $k=j+b$, class $e+3^js$, residue $\mu=e+3^js$ and bound $e+3^jB$. This yields $m\in\Tr(2w)$ with
\[
m\equiv e+3^js\pmod{3^{j+b}},\qquad m\equiv e+3^js\pmod M,\qquad m>e+3^jB .
\]
Reducing modulo $3^j$ shows $m=e+3^jt$ with $t>B$. By Lemma~\ref{lem:arith}(d), $t\equiv s\pmod{3^b}$ and $t\equiv s\pmod M$, hence $t\equiv s\pmod{3^bM}$.

Put $n=r'+2^{a'}t$. Then $n\ge t>B$, $n\equiv r'\pmod{2^{a'}}$ and $n\equiv r'+2^{a'}s\pmod{3^bM}$. By Lemma~\ref{lem:parity}, $T^{a'}(n)=e+3^jt=m$. By Lemmas~\ref{lem:shortcut} and~\ref{lem:tree_basic}(1), $n\reaches m\reaches2w\reaches w$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Write $N=2^a3^bM$ with $\gcd(M,6)=1$. By Lemma~\ref{lem:normalize} there are $a'\ge a$ and $r'\equiv r\pmod{2^a}$ with $j(a',r')\ge1$. For the odd modulus $Q=3^bM$, Lemma~\ref{lem:arith}(c) gives $s$ with $r'+2^{a'}s\equiv r\pmod Q$. Proposition~\ref{prop:construct} gives $n>B$ with $n\reaches w$, $n\equiv r'\equiv r\pmod{2^a}$ (since $a'\ge a$) and $n\equiv r'+2^{a'}s\equiv r\pmod Q$. As $\gcd(2^a,Q)=1$, Lemma~\ref{lem:arith}(b) gives $n\equiv r\pmod N$.
\end{proof}

\begin{remark}[Example]
Let $N=8$, $r=0$, $w=1$, $B=0$. Then $a=3$, $b=0$, $M=1$. Here $r_0=0$ and $T^3(0)=0$ is even, so Lemma~\ref{lem:normalize} takes $t=1$, $a'=4$, $r'=8$. The prefix $8,4,2,1$ has one odd term, so $j=1$, and $e=T^4(8)=2$. In $\Tr(2)$ the node $m=8$ satisfies $m\equiv e\pmod 3$ and $m>e$, so $t=(m-e)/3=2$ and $n=8+16\cdot2=40$. Indeed $40\equiv0\pmod8$ and $T^4(40)=8$ (the shortcut orbit is $40,20,10,5,8$), so $40\reaches8\reaches1$.
\end{remark}
